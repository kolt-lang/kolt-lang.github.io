Classes and Data Classes ​

Classes are top level only, and the body holds let, const, and fn declarations. Anything else belongs outside the class.

kolt
class Point {
  let x: int = 0
  const y: int = 0 // const keeps its default, init cannot reassign it
  fn init(x: int) {
    this.x = x;
  }
  fn toString(): string { // picked up by interpolation and string conversion
    return `Point(${this.x})`;
  }
}

let p = Point(10)
p.x = 99
// Point("oops") // error, constructor arity and arg types are checked
kolt
class NoInit {
  let x: int = 1
}
// no init written, automatic $init from field defaults fills it in
const n = NoInit()
io.println(n.x) // 1

Inheritance and super ​

kolt
class Derived extends Base {
  fn greet(): string {
    return super.greet(); // super calls only work inside a derived class
  }
  fn add(x: int, y: int): int {
    return super.add(x, y) * 2; // super.method<T>(x) takes type args too
  }
}

class C2<T> extends Base<T> {
  // dotted paths work too, type args flow through
}
class E2 extends Base<number> {
}

Private ​

kolt
// privacy is per class, not per file
class C {
  private let secret: string = "x"
  private fn helper(): int {
    return 1;
  }
}

data class P(private id: int, x: float) {
  // all fields come from the header
}

// C().secret // error, private fields stay inside the class

Data classes ​

Data classes put the fields in the header and skip the boilerplate:

kolt
// every field lives in the header, none in the body
data class Account(id: int, name: string = "a")
data class Secure(private secret: string, name: string)

// copy takes named args for public fields only
let a2 = a.copy(name = "Bob")
// a.copy(secret = "x") // error, unknown named parameter

Operator overloading ​

kolt
// plus, minus, times, slash, one param each
// each fires when the instance is on the left
class MathBox {
  let val: int
  fn init(v: int) {
    this.val = v;
  }
  fn plus(other: int): int {
    return this.val + other;
  }
  fn minus(other: int): int {
    return this.val - other;
  }
  fn times(other: int): int {
    return this.val * other;
  }
  fn slash(other: int): int {
    return this.val / other;
  }
}

let a = MathBox(10) + 2 // plus
let b = MathBox(10) - 2 // minus
let c = MathBox(10) * 2 // times
let d = MathBox(10) / 2 // slash

let e = MathBox(10)
e += 2 // plus again, then rebinds e to the int 12

Skipping an overload makes that operator fail with Method 'times' not found style errors, and a type mismatch points at the overload signature.

Enums ​

kolt
enum Color { // implicit 0, 1, 2
  Red,
  Green,
  Blue,
}
enum E { // explicit ints
  A = 1,
  B = 2,
}
enum N { // strings work too, never mix the two
  Alice = "a",
  Bob = "b",
}

let c: Color = Color.Red // member reads fold to literals at compile time
// Color() // error, enums are not callable
if x is Color {
  io.println("a color");
}

Computed members ​

kolt
enum Flags {
  Read = 1,
  Write = 2,
  Exec = 1 << 2, // shifts work, Exec is 4
  All = Read + Write + Exec, // earlier members by bare name, All is 7
  Mask = All & 7, // bitwise ops too, Mask is 7
}
io.println(Flags.All) // 7

kolt
