Error Handling ​

Throwing ​

Plain string ​

kolt
throw "boom"

Interpolated message ​

kolt
throw `Missing ${k}`

Structured error ​

kolt
throw error.create("E_X", "msg")

Propagating with try ​

Throwing function ​

kolt
fn read(p: string): string! {
  // try propagates the error to the caller
  return try fs.readText(p);
}

Handling with catch ​

Fallback value ​

kolt
let s = read("a") catch "fallback"

Fallback with the error ​

kolt
// intercepts locally, err holds the error
// separate operators at the same precedence
// try x catch y on one expression is rejected
let s2 = read("a") catch err {
  `bad:${err}`;
}

Throw contracts ​

Inferred automatically ​

kolt
fn risky(): int! {
  throw "x";
}
// no marker needed: calling a throwing function throws too
fn middle() {
  risky() // warns here, middle() callers warn as well
}

Always throws marker ​

kolt
// ! only documents the contract, inference works without it
fn boom2(): null! {
  throw "boom";
}

Conditional throws ​

kolt
// a throwing call outside try/catch warns, result widens to include Error
// rethrows only warns when the passed function can actually throw
fn myMap(arr: array<int>, f: fn(int): int): array<int> rethrows {
  let out: array<int> = []
  for x in arr {
    out.push(try f(x));
  }
  return out;
}

Error values ​

Read the message ​

kolt
risky() catch e {
  io.println(e.message());
}

Read the code ​

kolt
risky() catch e {
  io.println(e.code());
}

Read context and cause ​

kolt
risky() catch e {
  io.println(e.data());
}
risky() catch e {
  io.println(e.cause());
}

Defer ​

Stacked cleanup ​

kolt
// runs B then A, last in first out
fn f(): int {
  defer log.push("A")
  defer log.push("B")
  return 42;
}

IIFE form ​

kolt
let done = false
fn g() {
  defer (fn() {
    done = true;
  })();
}

Deferred error wins ​

kolt
// first deferred error replaces the return value
// bare values rejected, top level rejected, calls only, functions only
// runs per loop iteration and in workers before await completes
fn b(): int {
  defer throw "cleanup failed";
  return 1;
}
