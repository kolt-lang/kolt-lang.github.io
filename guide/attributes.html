Attributes and Reflection ​

Declarative metadata, validated at compile time and read back at runtime.

Declaring ​

Simple attribute ​

kolt
attribute Route(path: string, method: string = "GET")

Flag attribute ​

kolt
attribute Auth(required: bool)

Annotating ​

On a class ​

kolt
[Route("/users")]
[Auth(true)]
class UserController {
}

On a method ​

kolt
class UserController {
  [Route("/details", method = "POST")]
  fn getDetails(id: int): null {
  }
}

On fields and params ​

kolt
class Form {
  [Route("/save")] // fields take them too
  let action: string = ""
  fn submit([Auth(true)] verbose: bool) { // params take them too
  }
}
// dotted names like [ui.Json(...)] work
// args must be constant literals, counts, names, and visibility are checked

Sharing it ​

kolt
export attribute Route(path: string)

Reading back ​

All attributes ​

kolt
import std "reflect"

let attrs = try reflect.getAttributes(UserController)

Narrowing to the attribute type ​

kolt
for a in attrs {
  // narrows to Route here, so field access and completion just work
  if a is Route {
    io.println(a.path);
  }
}

Field attributes ​

kolt
try reflect.getFieldAttributes(UserController, "id")

Method attributes ​

kolt
try reflect.getMethodAttributes(UserController, "getDetails")

Member attributes ​

kolt
try reflect.getMemberAttributes(E, "Member")

Param attributes ​

kolt
try reflect.getParamAttributes(f, 0)

Serialization attributes ​

Rename a key ​

kolt
import std "reflect"

class U {
  // stdlib attributes need the module prefix, and sit on the field
  [reflect.JsonProperty("user_id")]
  let id: int = 0
}

Skip a field ​

kolt
import std "reflect"

class U {
  [reflect.Ignore] // left out of toMap and fromMap both ways
  let secret: string = ""
}

Round trip with comments on what happens ​

kolt
import std "reflect"

class User {
  [reflect.JsonProperty("user_id")] // map key becomes user_id
  let id: int = 0
  [reflect.Ignore] // never serialized
  let secret: string = ""
  let name: string = "anon" // no attribute, key stays name
}

const m = try reflect.toMap(User()) // {"user_id": 0, "name": "anon"}
const u2 = try reflect.fromMap(User, m) // id back from user_id, name keeps anon
// fromMap: missing required fields error with "Missing required field",
// field types are validated, ignored fields and fields with defaults keep
// their constructor values, [Alias("...")] names are tried after the primary key
