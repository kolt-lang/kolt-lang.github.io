Generics ​

Type params in <>, inferred from args or pinned with explicit type args.

Generic functions ​

Inference ​

kolt
fn id<T>(x: T): T {
  return x
}
io.println(id("hi")) // hi, T inferred as string

Explicit type args ​

kolt
// type arguments sit directly before (
io.println(id<int>(5)) // 5

Element typing ​

kolt
fn first<T>(arr: array<T>): T {
  return arr[0]
}
io.println(first([7, 8])) // 7

Bounds ​

Colon and extends ​

kolt
// both spellings bound T to numbers
fn clamp<T: number>(x: T, lo: T, hi: T): T {
  if x < lo {
    return lo
  }
  if x > hi {
    return hi
  }
  return x
}
io.println(clamp(3, 1, 10)) // 3
io.println(clamp(3.5, 1.0, 10.0)) // 3.5
// clamp("hi", "a", "z") // error, string does not satisfy number

Class bounds are nominal ​

kolt
class Shape {
  fn area(): float {
    return 0.0
  }
}
// must extend Shape, structural lookalikes do not count
class Circle extends Shape {
  let r: float = 1.0
  fn area(): float {
    return 3.14 * this.r * this.r
  }
}
fn total<T extends Shape>(shapes: array<T>): float {
  let sum = 0.0
  for s in shapes {
    sum += s.area()
  }
  return sum
}
io.println(total([Circle(), Circle()])) // 6.28

Sentinel bounds ​

kolt
// Instance matches any class instance
fn acceptInstance<T: Instance>(x: T): string {
  return "instance"
}
// fn matches any function
fn acceptFn<T: fn>(f: T): string {
  return "fn"
}
// Enum matches any enum definition object
enum E {
  A,
  B,
}
fn acceptEnum<T: Enum>(e: T): string {
  return "enum"
}
io.println(acceptInstance(Circle()))
io.println(acceptFn(id))
io.println(acceptEnum(E))

Generic classes ​

Concrete superclass arg ​

kolt
class Base<T> {
  let v: T
  fn init(x: T) {
    this.v = x
  }
}
class Sub extends Base<int> {
}
io.println(Sub(4).v) // 4

Passing the param through ​

kolt
class Wrap<T> extends Base<T> {
}
io.println(Wrap<string>("w").v) // w
