Language Server (LSP)

On this page

std net ​

Raw TCP, UDP, and DNS without HTTP framing. For HTTP use std http.

Full TCP ​

Server handlers receive each incoming chunk (a buffer, null on EOF), server owned connections read through the chunk argument, never through conn.read(). Clients use connect, write, and read with null narrowing for clean EOF.

kolt
import std "net"
import std "io"

// server: echo every chunk back, close on EOF
const srv = net.createServer((conn, chunk) => {
  if chunk is buffer {
    try conn.write(`hi ${chunk.toString()}`)
  } else {
    conn.close()
  }
})
// blocks serving, run it last
try srv.listen(9000)
kolt
import std "net"
import std "io"

// client: write, read the reply, print the peer
const c = try net.connect("127.0.0.1", 9000)
try c.write("ana")
const data = try c.read(64)
if data is not null {
  io.println(`tcp got: ${data.toString()}`)
}
io.println(`peer=${c.peerAddr()}`)
c.close()

Full UDP ​

kolt
import std "net"
import std "io"

const a = try net.bindUdp("127.0.0.1", 9001)
const b = try net.bindUdp("127.0.0.1", 9002)
try a.send("udp-hi", "127.0.0.1", 9002)
const pkt = try b.recv()
io.println(`udp got: ${pkt.data.toString()} from ${pkt.senderHost}:${pkt.senderPort}`)
io.println(`local=${try b.localAddr()}`)
a.close()
b.close()
