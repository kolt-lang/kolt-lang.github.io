std http ​

Full server ​

Routes, middleware, params, query strings, JSON bodies, forms, cookies, and a per-route auth chain. Handlers take (req, next) and next() moves to the next middleware or handler in the chain. Handlers that never call next() name it _, which the checker never registers.

kolt
import std "http"
import std "io"

const srv = http.server()

// global middleware runs before every route
srv.use((req, next) => {
  io.println(`${req.method()} ${req.path()}`)
  try next()
})

srv.get("/users/:id", (req, _) => {
  req.sendJson({id: req.param("id"), name: "ana"})
})

srv.post("/users", (req, _) => {
  const body = try req.json()
  req.sendJson({created: body}, 201)
})

srv.post("/form", (req, _) => {
  req.sendJson({name: req.form["name"] ?? "?", city: req.form["city"] ?? "?"})
})

srv.get("/login", (req, _) => {
  req.setCookie("session", "abc123", maxAge = 3600, httpOnly = true)
  req.sendJson({ok: true})
})

// route level handlers share the next() chain: auth first, then the real work
srv.get("/admin/stats",
  (req, next) => {
    if req.getQuery("token") != "42" {
      req.send("login required", 401)
    } else {
      try next()
    }
  },
  (req, _) => req.sendJson({users: 120, online: 8}))

srv.get("/old", (req, _) => req.redirect("/users/7"))
srv.get("/page", (req, _) => req.html("<h1>hi</h1>"))

// blocks serving, run it last
try srv.listen(8080)

Full client ​

Base URL, default headers, sticky cookie jar, JSON posting, batch fan-out, and status checks on the response (only transport failures throw, 4xx/5xx return normally).

kolt
import std "http"
import std "io"

const c = http.client("http://127.0.0.1:8080")
c.setHeader("Authorization", "Bearer x")

const me = try c.get("/users/7")
io.println(`status=${me.status}`)
const meBody = try me.json()
io.println(`body=${meBody}`)
io.println(`ok=${me.ok()}`)

const created = try c.postJson("/users", {name: "bob"})
const createdBody = try created.json()
io.println(`created=${createdBody}`)

const failed = try c.get("/admin/stats")
io.println(`rejected: ${failed.status}`)

// cookies set by the server land in the jar automatically
const login = try c.get("/login")
io.println(`session=${c.cookies["session"] ?? "none"}`)

// batch fan-out, one call for many requests
const out = try c.requestMany([
  http.HttpBatchJob("/users/7"),
  http.HttpBatchJob("/admin/stats?token=42", "GET"),
])
io.println(`batch got ${out.length()} results`)
