std stream ​

Chunked asynchronous and synchronous data pipelining with ReadableStream and WritableStream.

kolt
import std "stream"
import std "io"

// Create readable stream with items
const readable = stream.ReadableStream<string>()
try readable.push("chunk-1")
try readable.push("chunk-2")
readable.close()

// Create destination sink
const received: array<string> = []
const writable = stream.WritableStream<string>((chunk) => {
    received.push(chunk)
})

// Pipe data through
try stream.pipe<string>(readable, writable)
io.println(received) // ["chunk-1", "chunk-2"]

ReadableStream<T> methods ​

init(producer: (fn(): T?)? = null) ​

kolt
// Stream with passive queue
const r = stream.ReadableStream<string>()

// Stream with dynamic producer closure (returns null to signal EOF)
let counter = 0
const generator = stream.ReadableStream<int>(fn(): int? {
    if counter >= 3 { return null }
    counter++
    return counter
})

push(chunk: T): ReadableStream<T>! ​

kolt
try r.push("data-chunk")

Appends a chunk to the internal buffer. Throws if the stream is already closed.

read(): T?! ​

kolt
const chunk = try r.read()

Reads the next available chunk from the buffer or invokes the producer closure. Returns null on EOF.

isClosed(): bool ​

kolt
if r.isClosed() {
  io.println("Stream completed")
}

Returns true if the stream is closed and all buffered items have been consumed.

close() ​

kolt
r.close()

Closes the stream to new pushes.

pipe(dest: WritableStream<T>): WritableStream<T>! ​

kolt
try r.pipe(dest)

Streams all chunks into dest and calls dest.end() upon completion.

toArray(): array<T>! ​

kolt
const items = try r.toArray()

Drains all remaining items in the stream into an array.

WritableStream<T> methods ​

init(consumer: (fn(T))? = null) ​

kolt
const w = stream.WritableStream<string>((data) => {
    io.println(`Received: ${data}`)
})

Initializes a writable stream with an optional chunk handler.

write(chunk: T): bool! ​

kolt
try w.write("payload")

Writes a single chunk and triggers the consumer closure. Throws if closed.

writeAll(chunks: array<T>): bool! ​

kolt
try w.writeAll(["a", "b", "c"])

Writes each chunk in the array sequentially.

end() ​

kolt
w.end()

Closes the writable stream to further input.

isClosed(): bool ​

kolt
if w.isClosed() {
  io.println("Sink closed")
}

Returns true if end() has been called.

getWritten(): array<T> ​

kolt
const history = w.getWritten()

Returns an array of all chunks that have been written to this stream.

pipe<T>(readable: ReadableStream<T>, writable: WritableStream<T>): WritableStream<T>! ​

kolt
try stream.pipe<string>(src, dest)

Streams all available chunks from readable into writable, closing both upon completion.
