std ws ​

WebSocket client and server.

Full server ​

Echo server with connect, message, and close handlers. listen() pumps connections until stop().

kolt
import std "ws"
import std "io"

const srv = ws.server("127.0.0.1:8080")

srv.onConnect((c) => io.println("server: new conn"))

srv.onMessage((c, m) => {
  if m.isText() {
    try c.sendJson({echo: m.text()})
  }
  if m.isBinary() {
    try c.send(m.binary())
  }
})

srv.onClose((c, code, reason) => io.println(`server: closed ${code}`))

// blocks serving, run it last
try srv.listen()

Full client ​

Two styles: a manual read() loop, or handler callbacks with a listen() pump task.

kolt
import std "ws"
import std "io"

// style 1: read loop
const sock = try ws.connect("ws://127.0.0.1:8080/chat")
try sock.sendText("hello")
try sock.sendJson({kind: "join", room: "lobby"})
const msg = try sock.read()
if msg is not null {
  if msg.isText() {
    io.println(`got text: ${msg.text()}`)
  }
  if msg.isClose() {
    io.println(`server closed: ${msg.code}`)
  }
}
io.println(`open=${sock.isOpen()}`)
sock.close()
kolt
import std "ws"
import std "io"

// style 2: handlers plus a pump task
const sock = try ws.connect("ws://127.0.0.1:8080/chat")
sock.onMessage((m) => io.println(`handler got: ${m.text()}`))
sock.onClose((code, reason) => io.println(`closed: ${code}`))
sock.onError((e) => io.println(`error: ${e}`))
const pump = spawn try sock.listen()
try sock.sendText("ping")
sock.close()
await pump
