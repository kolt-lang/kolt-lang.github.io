std process ​

Run child processes, read env, and handle CLI args.

kolt
import std "process"
import std "io"

const st = process.exec("ls", ["-la"])
io.println(st.stdout)
io.println(process.getEnv("HOME"))
process.loadEnv() catch null // missing .env reads as null, no crash

Data classes ​

ProcessStatus(stdout: string, stderr: string, exitCode: int = 0) is what run and exec return. It also provides the success(): bool helper method.

kolt
const st = process.exec("echo", ["hi"])
io.println(st.stdout)
io.println(st.stderr)
io.println(st.exitCode)
io.println(st.success()) // true if exitCode == 0

Process methods ​

init(program: string, args: array<string> = []) ​

kolt
const p = process.Process("ls", ["-la"])

arg(str: string) ​

kolt
const p = process.Process("ls")
p.arg("-la").arg("/tmp") // chainable, returns this

args(arr: array<string>) ​

kolt
const p = process.Process("ls")
p.args(["-la", "/tmp"]) // pushes many, returns this

cwd(str: string) ​

kolt
const p = process.Process("ls")
p.cwd("/tmp") // sets working dir, returns this

silent(isSilent: bool = true) ​

kolt
const p = process.Process("ls")
p.silent() // does not echo the child's stdout/stderr to the console, returns this

run(): ProcessStatus ​

kolt
const st = process.Process("echo", ["hi"]).run()
io.println(st.stdout)

Standalone functions ​

exec(cmd: string, args: array<string> = []) ​

kolt
const st = process.exec("ls", ["-la"])
io.println(st.stdout)

One liner for build, run, and read.

exit(code: int = 0) ​

kolt
process.exit(1) // ends the process

pid(): int ​

kolt

cwd(): string ​

kolt

getEnv(name: string): string ​

kolt

kolt

hasEnv(name: string): bool ​

kolt
io.println(process.hasEnv("HOME")) // true when set

allEnv(): map<string, string> ​

kolt
const all = process.allEnv()
io.println(all["USER"])

setCwd(path: string)! ​

kolt
try process.setCwd("/tmp")

Changes the current working directory of the running process.

env(): map<string, string> ​

kolt
const envMap = process.env()
io.println(envMap["USER"])

Returns a copy of all current environment variables as a Kolt map.

args(): array<string> ​

kolt
io.println(process.args()) // CLI args after the file name

run(program: string, args: array<string> = []): ProcessStatus ​

kolt
const st = process.run("git", ["status"])
io.println(st.stdout)

loadEnv(path: string = ".env") ​

kolt
process.loadEnv() // .env in the working dir
process.loadEnv(".env.local")

Reads dotenv style files: trims quotes, skips blanks and # comments, sets each key.
