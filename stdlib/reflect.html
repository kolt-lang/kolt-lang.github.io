std reflect ​

Reflection pairs with the attribute system from the guide. Attributes carry the metadata, this module reads it back and helps with dynamic work.

kolt
import std "reflect"

class U {
  // qualified with the module name, sitting on the field itself
  [reflect.JsonProperty("user_id")]
  let id: int = 0
}

const m = try reflect.toMap(U()) // {"user_id": 0}
const u2 = try reflect.fromMap(U, m)

Standard attributes ​

kolt
import std "reflect"

class User {
  [reflect.JsonProperty("user_id")] // map key becomes user_id
  let id: int = 0
  [reflect.JsonProperty(required = true)] // must be present when reading back
  let email: string = ""
  [reflect.Optional] // may be missing, stays null
  let nickname: string? = null
  [reflect.Ignore] // never serialized either way
  let secret: string = ""
  let role: string = "user"
}

// narrowing works on attribute values too
const attrs = try reflect.getFieldAttributes(User(), "id")
for a in attrs {
  if a is reflect.JsonProperty {
    io.println(a.name)
  }
}

The rest of the standard set: Table(name) names the mapped table, Key(name = "") / Field(name = "") / Column(name = "") rename the primary key like JsonProperty, Alias(name = "") adds fallback names when reading back, Transient skips like Ignore.

toMap(instance: T): map! ​

kolt
import std "reflect"

class User {
  [reflect.JsonProperty("user_id")] // key becomes user_id
  let id: int = 0
  [reflect.Ignore] // skipped entirely
  let secret: string = ""
  let name: string = "anon"
}

const m = try reflect.toMap(User()) // {"user_id": 0, "name": "anon"}

fromMap(classRef: class<T>, data: map): T! ​

kolt
// constructor runs first, so defaults already exist
// user_id fills id, missing name keeps "anon"
// missing id with no default throws Missing required field
const u = try reflect.fromMap(User, {"user_id": 7})
io.println(u.id) // 7
io.println(u.name) // anon
