std url ​

WHATWG-compliant URL parsing and query parameter manipulation.

kolt
import std "url"
import std "io"

const parsed = try url.URL("https://user:pass@api.kolt.dev:8080/v1/users?sort=desc#profile")
io.println(parsed.hostname) // api.kolt.dev
io.println(parsed.pathname) // /v1/users
io.println(parsed.searchParams.get("sort")) // desc

URL methods ​

init(url: string, base: string? = null)! ​

kolt
const u = try url.URL("https://kolt.dev/docs")
const relative = try url.URL("/api", "https://kolt.dev")

Parses a URL string. Throws if the URL is invalid or malformed.

toString(): string ​

kolt
const u = try url.URL("https://kolt.dev/docs")
io.println(u.toString()) // https://kolt.dev/docs

Returns the full serialized URL string.

toJSON(): string ​

kolt
const u = try url.URL("https://kolt.dev/docs")
io.println(u.toJSON()) // https://kolt.dev/docs

Returns the serialized URL for JSON encoders.

URL fields ​

URL instances expose read-write URL components:

  • href: string: Full serialized URL
  • protocol: string: Protocol scheme including trailing : (e.g. https:)
  • username: string: User credential
  • password: string: Password credential
  • host: string: Host with port (e.g. api.kolt.dev:8080)
  • hostname: string: Host without port (e.g. api.kolt.dev)
  • port: string: Explicit port number or empty string
  • pathname: string: Path portion starting with /
  • search: string: Query string including leading ?
  • hash: string: Fragment identifier including leading #
  • origin: string: Origin tuple (scheme + host + port)
  • searchParams: UrlSearchParams: Query parameter manipulator

UrlSearchParams methods ​

init(init: string | map | array<(string, string)> = "") ​

kolt
const p1 = url.UrlSearchParams("q=kolt&lang=en")
const p2 = url.UrlSearchParams({"tag": "kolt", "limit": "10"})

get(name: string): string? ​

kolt
const p = url.UrlSearchParams("q=kolt")
io.println(p.get("q")) // kolt
io.println(p.get("missing")) // null

Returns the first value associated with name, or null if not found.

getAll(name: string): array<string> ​

kolt
const p = url.UrlSearchParams("tag=web&tag=cli")
io.println(p.getAll("tag")) // ["web", "cli"]

Returns all values matching name.

has(name: string): bool ​

kolt
const p = url.UrlSearchParams("q=kolt")
io.println(p.has("q")) // true

Checks whether name is present in the parameters.

set(name: string, value: string): UrlSearchParams ​

kolt
const p = url.UrlSearchParams("tag=old")
p.set("tag", "new")
io.println(p.toString()) // tag=new

Sets name to value, replacing all existing occurrences. Returns this for chaining.

append(name: string, value: string): UrlSearchParams ​

kolt
const p = url.UrlSearchParams("tag=first")
p.append("tag", "second")
io.println(p.toString()) // tag=first&tag=second

Appends a new name-value pair without overwriting existing entries.

delete(name: string): UrlSearchParams ​

kolt
const p = url.UrlSearchParams("tag=web&sort=asc")
p.delete("tag")
io.println(p.toString()) // sort=asc

Removes all query parameters matching name.

keys(): array<string> ​

kolt
const p = url.UrlSearchParams("a=1&b=2")
io.println(p.keys()) // ["a", "b"]

Returns an array of all parameter keys.

values(): array<string> ​

kolt
const p = url.UrlSearchParams("a=1&b=2")
io.println(p.values()) // ["1", "2"]

Returns an array of all parameter values.

entries(): array<(string, string)> ​

kolt
const p = url.UrlSearchParams("a=1&b=2")
for (k, v) in p.entries() {
  io.println(`${k}=${v}`)
}

Returns an array of key-value tuples.

size(): int ​

kolt
const p = url.UrlSearchParams("a=1&b=2")
io.println(p.size()) // 2

Returns the total count of parameter pairs.

isEmpty(): bool ​

kolt
const p = url.UrlSearchParams()
io.println(p.isEmpty()) // true

Returns true when no parameter pairs exist.

toString(): string ​

kolt
const p = url.UrlSearchParams({"name": "alice", "age": "30"})
io.println(p.toString()) // name=alice&age=30

Serializes the query parameters into a percent-encoded query string.
